📖 Code Examples & Integration

import shortcut from 'keysmaster';

// Initialize the global listener
shortcut.init();

// Cross-platform Cmd+S (Mac) / Ctrl+S (Windows)
const unbind = shortcut('mod+s', (e) => {
  e.preventDefault();
  console.log('Document saved!');
});

// Multi-key Vim / VS Code sequence
shortcut('ctrl+k ctrl+c', () => {
  console.log('Comment sequence triggered!');
});

// To unbind:
// unbind();
import { useState } from 'react';
import { useShortcut } from 'keysmaster/react';

export default function EditorComponent() {
  const [text, setText] = useState('');

  // Automatically binds on mount and unbinds on unmount
  useShortcut('mod+s', (e) => {
    console.log('Saved inside React component!');
  });

  // Modal close shortcut
  useShortcut('esc', () => {
    console.log('Escape pressed!');
  }, { scope: 'modal' });

  return <div>React Editor Content</div>;
}
import shortcut from 'keysmaster';

// Set active scope
shortcut.setScope('editor');

// Push nested scope (e.g. opening a modal dialog)

🧠 Smart `mod` Alias

Auto-resolves `mod+s` to ⌘ Cmd + S on macOS and Ctrl + S on Windows/Linux effortlessly.

🔀 Canonical Modifier Sorting

`shift+ctrl+a` and `ctrl+shift+a` evaluate identically with zero configuration needed.

🛡️ Form Input Filtering

Shortcuts are automatically blocked while typing in inputs unless explicitly enabled with `enableInInput: true`.

📚 Scope Stack & Fallback

Supports nested scope stacks for modals and editors while gracefully falling back to global shortcuts.

⚛️ Built-in React Hook

Includes exported `useShortcut()` hook with automatic setup and lifecycle unbinding.

🪶 Zero Dependencies

Extremely lightweight (< 10 KB unpacked bundle size), SSR-safe, and fully typed in TypeScript.
